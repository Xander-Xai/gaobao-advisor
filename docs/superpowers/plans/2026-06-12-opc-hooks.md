```python
# analytics/tracker.py
"""
轻量埋点模块 — 纯 Python + SQLite，零外部依赖。

隐私规则：
- 不记录用户输入原文
- 省份/专业/学校名是公开信息
- 情绪只记录档位，不记录触发词
- 无用户标识，只有匿名 session_id
"""

import sqlite3
import json
import os
from datetime import datetime, timedelta
from collections import Counter

_HERE = os.path.dirname(os.path.abspath(__file__))
_DEFAULT_DB = os.path.join(_HERE, "..", "data", "analytics.db")


class EventTracker:
    """轻量事件追踪器。"""

    def __init__(self, db_path=None):
        self.db_path = db_path or _DEFAULT_DB
        os.makedirs(os.path.dirname(self.db_path), exist_ok=True)
        self._conn = sqlite3.connect(self.db_path)
        self._conn.execute("""
            CREATE TABLE IF NOT EXISTS events (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                session_id TEXT NOT NULL,
                event_type TEXT NOT NULL,
                event_data TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        self._conn.execute("CREATE INDEX IF NOT EXISTS idx_events_session ON events(session_id)")
        self._conn.execute("CREATE INDEX IF NOT EXISTS idx_events_type ON events(event_type)")
        self._conn.execute("CREATE INDEX IF NOT EXISTS idx_events_time ON events(created_at)")
        self._conn.commit()

    def log_event(self, session_id: str, event_type: str, event_data: dict = None):
        """记录一个事件。"""
        data_str = json.dumps(event_data or {}, ensure_ascii=False)
        self._conn.execute(
            "INSERT INTO events (session_id, event_type, event_data) VALUES (?, ?, ?)",
            (session_id, event_type, data_str)
        )
        self._conn.commit()

    def get_stats(self, days: int = 7) -> dict:
        """获取最近 N 天的运营统计。"""
        cutoff = (datetime.now() - timedelta(days=days)).isoformat()
        cursor = self._conn.cursor()

        # 会话数
        cursor.execute(
            "SELECT COUNT(DISTINCT session_id) FROM events WHERE created_at >= ?",
            (cutoff,)
        )
        session_count = cursor.fetchone()[0]

        # 事件分布
        cursor.execute(
            "SELECT event_type, COUNT(*) FROM events WHERE created_at >= ? GROUP BY event_type",
            (cutoff,)
        )
        event_counts = dict(cursor.fetchall())

        # 情绪分布
        emotion_dist = {"🟢": 0, "🟡": 0, "🔴": 0}
        cursor.execute(
            "SELECT event_data FROM events WHERE event_type = 'emotion_detected' AND created_at >= ?",
            (cutoff,)
        )
        for row in cursor.fetchall():
            try:
                data = json.loads(row[0])
                level = data.get("level", "")
                if level in emotion_dist:
                    emotion_dist[level] += 1
            except (json.JSONDecodeError, TypeError):
                pass

        # 热门专业
        cursor.execute(
            "SELECT event_data FROM events WHERE event_type = 'query_major' AND created_at >= ?",
            (cutoff,)
        )
        major_counter = Counter()
        for row in cursor.fetchall():
            try:
                data = json.loads(row[0])
                name = data.get("major_name", "")
                if name:
                    major_counter[name] += 1
            except (json.JSONDecodeError, TypeError):
                pass

        # 热门学校
        cursor.execute(
            "SELECT event_data FROM events WHERE event_type = 'query_school' AND created_at >= ?",
            (cutoff,)
        )
        school_counter = Counter()
        for row in cursor.fetchall():
            try:
                data = json.loads(row[0])
                name = data.get("school_name", "")
                if name:
                    school_counter[name] += 1
            except (json.JSONDecodeError, TypeError):
                pass

        return {
            "session_count": session_count,
            "event_counts": event_counts,
            "emotion_distribution": emotion_dist,
            "top_majors": major_counter.most_common(10),
            "top_schools": school_counter.most_common(10),
            "days": days,
        }

    def close(self):
        """关闭数据库连接。"""
        self._conn.close()
```
